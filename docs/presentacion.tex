\section{Motivación y objetivo}

\begin{frame}{¿Por qué detección de anomalías?}
  \begin{block}{Limitación del aprendizaje supervisado en imagen médica}
    \begin{itemize}
      \item Necesita ejemplos anotados de cada alteración posible.
      \item La anotación radiológica requiere tiempo experto y presenta variabilidad.
      \item Una clase patológica no observada durante el entrenamiento puede quedar fuera.
    \end{itemize}
  \end{block}

  \vspace{0.35cm}
  \centering
  Aprender solo la \textbf{normalidad} y puntuar cuánto se aparta cada imagen:
  \[
    s(x_N)<s(x_A),\qquad x_N\sim p_N,\quad x_A\not\sim p_N.
  \]
\end{frame}

\begin{frame}{Pregunta de investigación}
  \begin{center}
  \begin{beamercolorbox}[rounded=true,shadow=true,sep=1em]{structure}
    \large ¿Puede un \textbf{denoising autoencoder}, entrenado solo con cortes
    FLAIR \textbf{normales}, separar de forma útil y reproducible los cortes
    normales de los anómalos en BraTS2021\_slice?
  \end{beamercolorbox}
  \end{center}

  \vspace{0.4cm}
  \begin{itemize}
    \item ``Anómalo'' significa no normal según BMAD; no confirma una patología.
    \item Objetivo: línea base sencilla, explicable y reproducible.
    \item Protocolo: pesos con train normal, umbral en validación, una evaluación en test.
  \end{itemize}
\end{frame}

\section{Fundamento del DAE}

\begin{frame}{Del autoencoder al denoising autoencoder}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{Autoencoder convencional}
      \begin{itemize}
        \item Aprende $x\rightarrow\hat{x}$.
        \item Un cuello de botella fuerte puede degradar tejido normal.
        \item Demasiada capacidad puede copiar también la anomalía.
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{DAE de Kascenas et al. \cite{kascenas2022denoising}}
      \begin{itemize}
        \item Aprende $\widetilde{x}\rightarrow x$.
        \item U-Net con conexiones de salto.
        \item Sin vector latente restrictivo.
        \item El ruido espacial obliga a aprender anatomía normal.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.35cm}
  \begin{alertblock}{Idea central}
    Una lesión hiperintensa no vista puede parecer una corrupción: el DAE tiende
    a retirarla o reconstruirla mal, dejando un residuo elevado.
  \end{alertblock}
\end{frame}

\begin{frame}{Ruido grueso durante el entrenamiento}
  \[
  n_s\sim\mathcal N(0,I)_{16\times16},\qquad
  \widetilde{x}=x+0{,}2\,M\odot\operatorname{roll}(\operatorname{up}(n_s))
  \]

  \begin{enumerate}
    \item Muestrear ruido gaussiano a $16\times16$.
    \item Interpolarlo bilinealmente a $128\times128$.
    \item Trasladarlo aleatoriamente para variar los centros de los patrones.
    \item Aplicarlo solo al cerebro: $M=\mathbb{1}(x>0{,}01)$.
  \end{enumerate}

  \begin{block}{Objetivo}
    \[
      \mathcal L=\frac{\sum M\odot(f_\theta(\widetilde{x})-x)^2}{\max(\sum M,1)}
    \]
    MSE únicamente sobre el primer plano; el fondo no domina el aprendizaje.
  \end{block}
\end{frame}

\begin{itemize}
  \item Cada bloque: $[\mathrm{Conv}_{3\times3}\rightarrow\mathrm{SiLU}
        \rightarrow\mathrm{GroupNorm}]\times2$.
  \item Average pooling; subida bilineal seguida de convolución; head $1\times1$.

\begin{frame}{Por qué estas decisiones}
  \begin{tabular}{p{3.6cm}p{7.8cm}}
    \toprule
    Decisión & Función \\
    \midrule
    Convolución $3\times3$ & Extraer contexto local conservando resolución \\
    SiLU/Swish & Activación suave utilizada en el método de referencia \\
    GroupNorm & Estabilidad sin depender de lotes grandes \\
    Average pooling & Reducir resolución preservando intensidades suaves \\
    Skip connections & Recuperar bordes y detalle anatómico normal \\
    Salida lineal & Regresión de intensidad sin saturación de una sigmoide \\
    \bottomrule
  \end{tabular}
\end{frame}

\section{Score y adaptación del paper}

\begin{frame}{De la reconstrucción al score por corte}
  En test se introduce la imagen \textbf{limpia}; no se añade ruido.
  \[
    A(x)=f_{\mathrm{med},5}\left(
    \widetilde M\odot|x-f_\theta(x)|\right),\qquad
    s(x)=\max_{i,j}A(x)_{ij}.
  \]

  \begin{enumerate}
    \item Erosionar la máscara para ignorar errores del borde del cráneo.
    \item Calcular error absoluto por píxel.
    \item Aplicar mediana $5\times5$ para eliminar respuestas aisladas.
    \item Tomar el máximo espacial: una puntuación por corte.
  \end{enumerate}

  \begin{alertblock}{Diferencia clave}
    El paper conserva $A(x)$ para segmentar; BMAD exige reducirlo a $s(x)$ para
    clasificar toda la imagen.
  \end{alertblock}
\end{frame}

\begin{frame}{Qué conservamos y qué adaptamos}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \textbf{Conservado del paper}
      \begin{itemize}
        \item U-Net, 3 reducciones y skips.
        \item Canales 64--128--256--512.
        \item SiLU, GroupNorm y average pooling.
        \item Ruido $16\times16$, $\sigma=0{,}2$ y roll.
        \item MSE de primer plano.
        \item Residuo absoluto + mediana $5\times5$.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Adaptado al proyecto}
      \begin{itemize}
        \item 1 FLAIR en lugar de 4 modalidades.
        \item Clasificación por corte, no segmentación.
        \item Convolución estándar, sin weight standardization.
        \item Bilinear + conv en lugar de transposed conv.
        \item 8 grupos fijos y padding cero.
        \item Normalización del PNG a $[0,1]$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Por qué cambia la evaluación}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    & Paper & Este proyecto \\
    \midrule
    Dato & 4 modalidades + máscara & FLAIR + etiqueta de corte \\
    Salida evaluada & mapa por píxel & score escalar \\
    Métricas & AUPRC y Dice & AUROC y balanced accuracy \\
    Umbral & Dice óptimo analítico & Youden en validación \\
    Postproceso & mediana y CC 3D & mediana 2D \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.35cm}
  \textbf{Razón metodológica}: no inventar máscaras ni estructura 3D ausentes y
  no utilizar test para calibrar la decisión.
\end{frame}

\section{Entrenamiento y evaluación}

\begin{frame}{Configuración reproducible}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Entrenamiento}
      \begin{itemize}
        \item Batch: 16.
        \item Adam + AMSGrad.
        \item $\mathrm{lr}_{\max}=10^{-4}$.
        \item Weight decay: $10^{-5}$.
        \item Ciclo coseno: 200 updates.
        \item 143 épocas $\approx$ 67.067 updates.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Control experimental}
      \begin{itemize}
        \item Semillas: 13, 42 y 73.
        \item cuDNN determinista.
        \item Mejor época por MSE normal de validación.
        \item Umbral fijado con validación etiquetada.
        \item Test evaluado sin reajustes.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Métricas}
  \begin{block}{AUROC}
    Probabilidad de que una imagen anómala elegida al azar reciba mayor score
    que una normal. No depende del umbral; $0{,}5$ equivale al azar.
  \end{block}

  \begin{block}{Balanced accuracy}
    \[
      \mathrm{BA}=\frac{1}{2}
      \left(\frac{TP}{TP+FN}+\frac{TN}{TN+FP}\right).
    \]
    Da el mismo peso a normales y anómalas pese al desbalance de test.
  \end{block}

  Se reportará \textbf{media $\pm$ desviación típica} de las tres semillas.
\end{frame}

\section{Reproducibilidad y conclusiones}

\begin{frame}{Entregables reproducibles}
  \begin{itemize}
    \item Configuración, historial completo y época seleccionada en \texttt{metrics.json}.
    \item Pesos y arquitectura en \texttt{model.pt}.
    \item Scores individuales de validación y test en CSV.
    \item Reconstrucciones visuales de control.
    \item Notebook de Colab con comprobación de GPU, smoke test y campaña completa.
    \item Ejecución final de tres semillas y resumen agregado en
          \texttt{results/resultados.csv}.
  \end{itemize}
\end{frame}

\begin{frame}{Limitaciones}
  \begin{itemize}
    \item Una modalidad 2D pierde información multimodal y tridimensional.
    \item La etiqueta por corte no permite evaluar localización tumoral.
    \item El máximo espacial puede reaccionar a artefactos locales.
    \item La validación es pequeña: 39 normales y 44 anómalas.
    \item Un benchmark no demuestra generalización a otros centros o escáneres.
    \item Software experimental; no apto para diagnóstico ni decisiones clínicas.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item Se implementó una adaptación unimodal y por corte del DAE de
          Kascenas et al., entrenada \textbf{solo con normales}.
    \item La red conserva el núcleo del método: U-Net sin bottleneck restrictivo,
          ruido grueso y error de reconstrucción filtrado.
    \item Entrenamiento, selección de época, calibración y test están separados.
    \item La campaña de tres semillas permitirá cuantificar rendimiento y variabilidad.
    \item Las conclusiones cuantitativas quedan pendientes de los resultados de Colab.
  \end{itemize}

  \vspace{0.25cm}
  \textbf{Trabajo futuro}: recuperar cuatro modalidades y máscaras, evaluar por
  paciente y validar externamente.
\end{frame}

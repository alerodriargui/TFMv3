\section{Problema y objetivo}

\begin{frame}{¿Qué problema intento resolver?}
  \begin{block}{Planteamiento}
    En lugar de enseñar al modelo todas las patologías posibles, se aprende
    únicamente cómo son los cortes \textbf{normales}.
  \end{block}

  \vspace{0.25cm}
  \centering
  \[
    s(x_N)<s(x_A),\qquad x_N\sim p_N,\quad x_A\not\sim p_N.
  \]

  \begin{itemize}
    \item Los pesos se entrenan solo con \texttt{train/good} (imágenes normales).
    \item Una anomalía es una imagen incompatible con la distribución normal aprendida.
    \item El sistema no identifica una enfermedad concreta, eso requeriría un enfoque supervisado.
  \end{itemize}
\end{frame}

\begin{frame}{Pregunta de investigación}
  \begin{center}
  \begin{beamercolorbox}[rounded=true,shadow=true,sep=1em]{structure}
    \large ¿Puede un \textbf{denoising autoencoder}, entrenado solo con cortes
    FLAIR \textbf{normales}, separar cortes normales de anómalos en
    BraTS2021\_slice mediante el error de reconstrucción?
  \end{beamercolorbox}
  \end{center}

  \vspace{0.35cm}
\begin{itemize}
    \item \textbf{Entrenamiento del modelo:} aprendizaje de los pesos utilizando únicamente imágenes normales.
    \item \textbf{Selección del modelo:} elección de la mejor época mediante el conjunto de validación normal.
    \item \textbf{Calibración del umbral:} determinación del umbral de decisión utilizando imágenes normales y anómalas de validación.
    \item \textbf{Evaluación final:} evaluación sobre el conjunto de prueba, sin reajustar los pesos del modelo ni el umbral de decisión.
\end{itemize}
\end{frame}

\begin{frame}{¿Por qué una máscara de primer plano?}
  El fondo de la resonancia es prácticamente negro y no aporta información útil
  para detectar anomalías cerebrales.

  \[
    M=\mathbb{1}(x>0{,}01)
  \]

  \begin{itemize}
    \item El umbral 0,01 separa fondo casi negro de primer plano.
    \item El ruido de entrenamiento se aplica solo donde $M=1$.
    \item La MSE también se calcula solo sobre el primer plano.
    \item Así el modelo no obtiene una pérdida artificialmente baja por reconstruir fondo negro.
  \end{itemize}
\end{frame}

\section{Del autoencoder al DAE}

\begin{frame}{Primero: ¿qué es un autoencoder (AE)?}
  Un \textbf{autoencoder} es una red neuronal que aprende a reproducir su entrada
  sin necesitar una etiqueta de clase.
  \[
    x\xrightarrow[\text{comprime}]{\text{encoder }g_\phi}
    z\xrightarrow[\text{reconstruye}]{\text{decoder }h_\psi}
    \hat{x},\qquad f_\theta=h_\psi\circ g_\phi.
  \]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Encoder}
      \begin{itemize}
        \item Extrae características relevantes.
        \item Reduce progresivamente la resolución.
        \item Aumenta el contexto espacial de cada activación.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Decoder}
      \begin{itemize}
        \item Recupera la resolución original.
        \item Transforma las características aprendidas en la imagen reconstruida.
        \item Produce $\hat{x}$ con el mismo tamaño que $x$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Entonces, ¿qué es un DAE y qué hace?}
  Un \textbf{denoising autoencoder (DAE)} es un AE entrenado para recuperar una
  imagen limpia $x$ a partir de una versión corrompida $\widetilde{x}$:
  \[
    x\xrightarrow{q(\widetilde{x}\mid x)}\widetilde{x}
    \xrightarrow{f_\theta}\hat{x}\approx x.
  \]

  \begin{enumerate}
    \item Toma un corte cerebral normal y añade una corrupción artificial.
    \item El encoder identifica estructura y contexto que sobreviven al ruido.
    \item El decoder restaura una versión limpia plausible.
    \item Para lograrlo, aprende patrones de los cortes normales, no una copia píxel a píxel.
  \end{enumerate}

  \begin{block}{Diferencia esencial}
    AE: $x\mapsto x$. DAE: $\widetilde{x}\mapsto x$. La etiqueta objetivo sigue
    siendo la propia imagen, pero la entrada y el objetivo ya no son idénticos.
  \end{block}
\end{frame}

\begin{frame}{¿Por qué un DAE y no copiar la entrada?}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{Autoencoder convencional}
      \[
        x\longrightarrow f_\theta(x)\approx x
      \]
      \begin{itemize}
        \item Puede aproximarse demasiado a la identidad.
        \item Una red potente podría reconstruir también anomalías.
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Denoising Autoencoder}
      \[
        \widetilde{x}\longrightarrow f_\theta(\widetilde{x})\approx x
      \]
      \begin{itemize}
        \item La entrada se corrompe artificialmente.
        \item El objetivo es recuperar la anatomía normal limpia.
        \item La tarea obliga a utilizar contexto espacial.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  Basado en la idea de Kascenas et al. \cite{kascenas2022denoising}.
\end{frame}

\begin{frame}{Entrenamiento e inferencia no hacen lo mismo}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Entrenamiento}
      \[
        \widetilde{x}=x+n\longrightarrow \hat{x},
        \qquad \mathcal L(\hat{x},x)
      \]
      \begin{itemize}
        \item Solo imágenes normales.
        \item Se añade ruido nuevo en cada pasada.
        \item Se actualizan los pesos para recuperar $x$.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Inferencia}
      \[
        x\longrightarrow\hat{x},
        \qquad A(x)=|x-\hat{x}|
      \]
      \begin{itemize}
        \item Se introduce la imagen limpia.
        \item No se actualizan pesos ni se añade ruido.
        \item El residuo se convierte en score de anomalía.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ruido grueso durante el entrenamiento}
  \[
  n_s\sim\mathcal N(0,I)_{16\times16},\qquad
  \widetilde{x}=x+0{,}2\,M\odot\operatorname{roll}(\operatorname{up}(n_s))
  \]

  \begin{enumerate}
    \item Generar ruido gaussiano a baja resolución: $16\times16$.
    \item Interpolarlo bilinealmente a $128\times128$.
    \item Desplazarlo circularmente en posiciones aleatorias.
    \item Aplicarlo solo al primer plano cerebral.
  \end{enumerate}

  \begin{block}{Por qué ruido grueso}
    Produce perturbaciones espaciales suaves y de mayor escala; eliminar ruido
    píxel a píxel sería una tarea más local y potencialmente más trivial.
  \end{block}
\end{frame}

\begin{frame}{Función de pérdida}
  La red reconstruye la imagen limpia:
  \[
    \hat{x}=f_\theta(\widetilde{x})
  \]

  y optimiza una MSE de primer plano:
  \[
    \mathcal L(\theta)=
    \frac{\sum M\odot(\hat{x}-x)^2}{\max(\sum M,1)}.
  \]

  \begin{itemize}
    \item Es una regresión píxel a píxel.
    \item El cuadrado penaliza especialmente errores grandes.
    \item El fondo se excluye para centrar el aprendizaje en tejido cerebral.
  \end{itemize}
\end{frame}

\begin{frame}{Arquitectura U-Net del DAE}

  \draw[->,dashed,blue,bend left=36] (e1) to node[below]{skip} (d1);
  \draw[->,dashed,blue,bend left=28] (e2) to node[below]{skip} (d2);

\vspace{0.4cm}
\begin{itemize}
  \item Cuatro niveles: 64, 128, 256 y 512 canales.
  \item Tres reducciones por \texttt{AvgPool2d(2)}.
  \item Tres subidas bilineales + convolución.
  \item Skip connections entre escalas equivalentes.
  \item Salida final $1\times128\times128$ mediante convolución $1\times1$.
  \item \textbf{8.558.721 parámetros entrenables}.
\end{itemize}
\end{frame}

\begin{frame}{Qué hace cada bloque}
  Cada \texttt{ConvBlock} aplica dos veces:
  \[
    \mathrm{Conv}_{3\times3}\rightarrow\mathrm{SiLU}\rightarrow\mathrm{GroupNorm}(8).
  \]

  \begin{tabular}{p{3.6cm}p{7.6cm}}
    \toprule
    Elemento & Motivo \\
    \midrule
    Conv. $3\times3$ & Extraer patrones espaciales locales con pocos parámetros \\
    \texttt{padding=1} & Mantener altura y anchura dentro de cada bloque \\
    SiLU & Introducir una no linealidad suave \\
    GroupNorm & Normalizar sin depender de estadísticas entre ejemplos del batch \\
    Average pooling & Reducir resolución de forma suave \\
    Skip connections & Recuperar detalle espacial para la reconstrucción \\
    Salida lineal & Regresión de intensidad sin saturación de una sigmoide \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{¿Qué ocurre en una skip connection?}
  Ejemplo en la primera etapa del decoder:
  \[
    d_3=\mathrm{dec3}\left(
      \operatorname{cat}(\mathrm{up3}(e_4),e_3)
    \right).
  \]

  \begin{itemize}
    \item $e_4$: representación profunda a $16\times16\times512$.
    \item \texttt{up3}: la convierte en $32\times32\times256$.
    \item $e_3$: conserva detalle del encoder a $32\times32\times256$.
    \item La concatenación se hace en canales: $256+256=512$.
    \item \texttt{dec3} transforma esos 512 canales en 256.
  \end{itemize}

  \begin{block}{Idea}
    El decoder combina contexto profundo con información espacial de alta resolución.
  \end{block}
\end{frame}

\begin{frame}{¿Por qué una U-Net para un autoencoder?}
  \begin{itemize}
    \item La \textbf{rama descendente} aumenta el campo receptivo: permite usar
          contexto anatómico y no solo comparar píxeles aislados.
    \item La \textbf{rama ascendente} devuelve la representación a la resolución
          necesaria para producir un mapa de error espacial.
    \item Las \textbf{skip connections} recuperan bordes y detalle que podrían
          perderse al reducir resolución.
    \item Los canales 64--128--256--512 aumentan la capacidad al disminuir el
          tamaño espacial, donde las activaciones son más baratas.
  \end{itemize}

  \begin{alertblock}{¿Las skips no facilitan copiar la anomalía?}
    Sí, es un riesgo real. Se conservan porque la reconstrucción médica necesita
    precisión espacial; el entrenamiento con corrupción gruesa obliga a combinar
    esas características locales con contexto. Es un compromiso, no una garantía.
  \end{alertblock}
\end{frame}

\begin{frame}{Justificación de las operaciones arquitectónicas}
  \small
  \begin{tabular}{p{3.0cm}p{8.8cm}}
    \toprule
    Elección & Justificación \\
    \midrule
    Dos conv. $3\times3$ & Amplían el campo receptivo con menos parámetros que una convolución grande y permiten dos transformaciones no lineales \\
    SiLU & Activación suave y derivable, con gradiente también para entradas negativas \\
    GroupNorm (8 grupos) & No usa estadísticas del lote; es estable con batch 16 y divide exactamente todos los anchos de canal \\
    Average pooling & Reduce resolución suavizando, en lugar de escoger solo el máximo local \\
    Bilinear + conv. & Separa ampliación y aprendizaje y evita patrones de tablero asociados a algunas convoluciones transpuestas \\
    Cabeza $1\times1$ lineal & Combina los 64 canales en una intensidad sin añadir contexto ni saturar la salida \\
    \bottomrule
  \end{tabular}
\end{frame}

\begin{frame}{Adam y scheduler}
  Para cada lote:
  \[
    \texttt{zero\_grad}\rightarrow
    \texttt{forward}\rightarrow
    \mathcal L\rightarrow
    \texttt{backward}\rightarrow
    \texttt{optimizer.step}.
  \]

  \begin{itemize}
    \item \textbf{Adam}: adapta el tamaño de actualización usando momentos del gradiente.
    \item \textbf{AMSGrad}: variante de Adam que conserva un máximo histórico del segundo momento.
    \item \textbf{Weight decay}: regulariza los pesos.
    \item \textbf{CosineAnnealingLR}: $T_{\max}=100$.
  \end{itemize}

  \begin{alertblock}{Detalle del código actual}
    \texttt{scheduler.step()} se ejecuta por lote, por lo que $T_{\max}=100$
    está expresado en \textbf{actualizaciones}, no en épocas. Esto produce ciclos
    rápidos de la tasa y debe revisarse mediante una ablación, no presentarse
    como una ventaja demostrada.
  \end{alertblock}
\end{frame}

\begin{frame}{¿Por qué Adam, AMSGrad y reproducibilidad?}
  \begin{itemize}
    \item \textbf{Adam} adapta el paso de cada parámetro usando medias móviles del
          gradiente y su cuadrado; ayuda cuando distintas capas tienen escalas
          de gradiente diferentes.
    \item \textbf{AMSGrad} conserva el máximo histórico del segundo momento; se
          adopta como variante conservadora de Adam.
    \item La \textbf{semilla 42} fija Python, NumPy, PyTorch y CUDA.
    \item \textbf{cuDNN determinista} y \texttt{benchmark=False} priorizan
          repetibilidad frente a velocidad máxima.
  \end{itemize}

  \begin{alertblock}{Límite de la semilla}
    Una ejecución reproducible no mide variabilidad. Para afirmar robustez hacen
    falta varias semillas y media $\pm$ desviación estándar.
  \end{alertblock}
\end{frame}

\begin{frame}{Selección del mejor modelo}
  Después de cada época se evalúan únicamente imágenes normales de validación.

  \[
    e^*=\arg\min_e\;\mathrm{MSE}_{\mathrm{valid/good}}(e)
  \]

  \begin{itemize}
    \item Se entrenan las 50 épocas.
    \item Cuando mejora la MSE normal de validación se guarda una copia de los pesos.
    \item Al terminar, se recargan los pesos de la mejor época.
    \item Las imágenes anómalas no participan en esta selección.
  \end{itemize}
\end{frame}

\section{Inferencia y score}

\begin{frame}{De la reconstrucción al mapa de anomalía}
  En inferencia se introduce la imagen \textbf{limpia}; no se añade ruido.
  \[
    e(x)=|x-f_\theta(x)|
  \]

  Después:
  \begin{enumerate}
    \item Refinar la máscara con average pooling $5\times5$ y umbral 0,95.
    \item Eliminar el error fuera del interior cerebral.
    \item Aplicar mediana local $5\times5$.
  \end{enumerate}

  \[
    A(x)=f_{\mathrm{med},5}\left(
      \widetilde M\odot|x-f_\theta(x)|
    \right).
  \]

  El filtro de mediana reduce respuestas puntuales aisladas.
\end{frame}

\begin{frame}{¿Por qué el máximo como score?}
  BMAD proporciona una etiqueta por corte, por lo que el mapa debe reducirse a
  un único número:
  \[
    s(x)=\max_{i,j}A(x)_{ij}.
  \]

  \begin{itemize}
    \item Una anomalía puede ocupar una región pequeña.
    \item Un promedio global podría diluirla entre miles de píxeles normales.
    \item El máximo pregunta si existe alguna región con error especialmente alto.
    \item La mediana previa evita que un único píxel ruidoso domine el score.
  \end{itemize}
\end{frame}

\begin{frame}{¿Por qué error absoluto, máscara y mediana?}
  \small
  \begin{tabular}{p{3.1cm}p{5.6cm}p{3.1cm}}
    \toprule
    Operación & Función & Riesgo controlado \\
    \midrule
    $|x-\hat{x}|$ & Mide magnitud del residuo sin cancelar signos & Una diferencia no implica por sí sola patología \\
    Máscara erosionada & Exige que casi todo el vecindario $5\times5$ sea cerebro & Bordes cráneo--fondo producen errores altos \\
    Mediana $5\times5$ & Conserva respuestas locales consistentes y elimina impulsos & Un píxel espurio dominaría el máximo \\
    Máximo espacial & Detecta una región pequeña sin diluirla & Puede favorecer un artefacto local persistente \\
    \bottomrule
  \end{tabular}

  \vspace{0.25cm}
  \begin{block}{Por qué no MSE como mapa de inferencia}
    El error absoluto mantiene el residuo en la escala de intensidad y crece
    linealmente; la MSE se reserva como objetivo de entrenamiento, donde interesa
    penalizar con fuerza reconstrucciones muy alejadas.
  \end{block}
\end{frame}

\section{Calibración y evaluación}

\begin{frame}{Selección del umbral}
  Con los scores y etiquetas de \textbf{validación} se calcula la curva ROC.
  El código selecciona:
  \[
    t^*=\arg\max_t\left(TPR(t)-FPR(t)\right).
  \]

  Es el criterio de Youden:
  \[
    J=\mathrm{sensibilidad}+\mathrm{especificidad}-1.
  \]

  \begin{block}{Regla metodológica}
    El threshold se selecciona una sola vez en validación y después se aplica
    sin cambios al test.
  \end{block}
\end{frame}

\begin{frame}{¿Por qué Youden y por qué nunca calibrar en test?}
  \begin{itemize}
    \item Youden maximiza $TPR-FPR$, equivalente a maximizar
          sensibilidad $+$ especificidad $-1$.
    \item Da el mismo peso a ambas clases y no depende de su proporción en el
          conjunto, algo importante con clases desbalanceadas.
    \item Produce una regla binaria única sin imponer aquí un coste clínico
          específico para FP o FN.
    \item Calibrar en test filtraría información del conjunto reservado y daría
          una estimación optimista del rendimiento.
  \end{itemize}

  \begin{alertblock}{Qué no significa}
    No es necesariamente el umbral clínico óptimo. Si el coste de un falso
    negativo fuese mayor, habría que elegir otro punto operativo y validarlo
    prospectivamente.
  \end{alertblock}
\end{frame}

\begin{frame}{Métricas}
  \begin{block}{AUROC}
    Evalúa la capacidad del score para ordenar anómalas por encima de normales
    sin depender de un threshold concreto. $0{,}5$ equivale al azar y 1 a
    separación perfecta.
  \end{block}

  \begin{block}{Balanced accuracy}
    \[
      \mathrm{BA}=\frac{1}{2}
      \left(\frac{TP}{TP+FN}+\frac{TN}{TN+FP}\right).
    \]
    Es adecuada porque el test contiene 640 normales y 3.075 anómalas.
  \end{block}
\end{frame}

\begin{frame}{¿Por qué AUROC y balanced accuracy juntas?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{AUROC: calidad del ranking}
      \begin{itemize}
        \item Usa todos los umbrales posibles.
        \item No depende del threshold seleccionado.
        \item Mide si una anómala tiende a recibir mayor score que una normal.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{BA: decisión operativa}
      \begin{itemize}
        \item Evalúa el threshold ya congelado.
        \item Promedia sensibilidad y especificidad.
        \item Evita que las 3.075 anómalas oculten el rendimiento en 640 normales.
      \end{itemize}
    \end{column}
  \end{columns}

  \begin{block}{Por qué no accuracy convencional}
    Predecir siempre la clase mayoritaria ya daría $3.075/3.715=82{,}8\%$ de
    accuracy, aunque la especificidad fuese cero.
  \end{block}
\end{frame}

        \item DAE tipo U-Net.
        \item Tres reducciones y skips.
        \item Ruido gaussiano grueso.
        \item $16\times16$ y $\sigma=0{,}2$.

    \begin{column}{0.49\textwidth}
      \textbf{Adaptaciones del proyecto}
      \begin{itemize}
        \item Un canal FLAIR.
        \item Cortes 2D independientes.
        \item Score escalar por imagen.
        \item Interpolación bilineal + convolución.
        \item GroupNorm con 8 grupos.
        \item AUROC y balanced accuracy.
        \item Threshold de Youden en validación.

\begin{frame}{¿Cómo interpretar los resultados?}
  \begin{itemize}
    \item AUROC 0,9633: el score ordena correctamente una pareja
          anómala--normal elegida al azar en aproximadamente el 96,3\% de los casos.
    \item Sensibilidad 0,9210: detecta 2.832 de 3.075 cortes anómalos.
    \item Especificidad 0,8984: identifica como normales 575 de 640 cortes normales.
    \item Balanced accuracy 0,9097: promedio de las dos tasas anteriores.
    \item La mejora de AUROC de validación (0,9452) a test (0,9633) puede ser
          variación muestral; no demuestra que el modelo mejore al ver test.
  \end{itemize}

  \begin{alertblock}{Conclusión proporcionada}
    Los resultados apoyan la separabilidad en este benchmark y esta ejecución;
    no prueban diagnóstico, localización exacta ni generalización clínica.
  \end{alertblock}
\end{frame}

    \midrule

\begin{frame}{Limitaciones}
  \begin{itemize}
    \item Una modalidad FLAIR 2D pierde información multimodal y tridimensional.
    \item Una etiqueta por corte no permite evaluar localización de lesión.
    \item El máximo espacial puede ser sensible a artefactos locales.
    \item La validación es pequeña: 39 normales y 44 anómalas.
    \item El threshold usa etiquetas de validación: los pesos son no supervisados,
          pero la decisión binaria final está calibrada supervisadamente.
    \item Un benchmark no demuestra generalización clínica externa.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item Se implementó un DAE tipo U-Net entrenado \textbf{solo con normales}.
    \item El ruido grueso convierte el aprendizaje en una tarea de restauración,
          no en una simple copia de la entrada.
    \item El error de reconstrucción se filtra y reduce a un score por corte.
    \item Pesos, calibración y test están metodológicamente separados.
    \item En test: AUROC 0,9633 y balanced accuracy 0,9097.
  \end{itemize}

  \vspace{0.25cm}
  \textbf{Trabajo futuro}: varias semillas, ablaciones de hiperparámetros,
  información multimodal/3D, máscaras de lesión y validación externa.
\end{frame}
